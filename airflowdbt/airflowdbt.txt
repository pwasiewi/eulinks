##### DBT postgres
```
pip3 install --user dbt-core dbt-postgres dbt-redshift dbt-snowflake
dbt init dbt
cd dbt
mkdir profile
# edit profile/profiles.yml
local:
  target: dev
  outputs:
    dev:
      type: postgres
      threads: 1
      host: localhost
      port: 5432
      user: postgres
      pass: ""
      dbname: postgres
      schema: public
export DBT_PROFILES_DIR=`pwd`/profile
# edit dbt_project.yml - change to profile: 'local'
dbt debug
```
